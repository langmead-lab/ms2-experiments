% Fig Threads (appendix): thread scaling on the HPRC2 1 Gbase short-read and Nanopore sets.
% Left pair: speed (Mbases/s of query time) against threads; right pair: speedup over one
% thread, with the dashed line for linear scaling up to the 32 physical cores. Threads 1 to 32 run on physical cores of one
% socket; 64 runs on both hardware threads of those 32 cores, the shaded band, labeled Hyperthreading (SMT). Data from
% fig_threads_data.py.
\documentclass[border=1pt]{standalone}
\input{figures-style}
\input{figures-tweaks}
\begin{document}
\pgfplotsset{
  ms link/rl1/.style={fig link=ropeRed},
  threads axis/.style={fig axis, xmin=0.8, xmax=128,
    xtick={1,2,4,8,16,32,64}, xticklabels={1,2,4,8,16,32,64}},
  speed axis/.style={threads axis, ymin=0.4, ymax=1500,
    ytick={1,10,100,1000}, yticklabels={1,10,100,1000}},
  speedup axis/.style={threads axis, ymin=0.8, ymax=80,
    ytick={1,2,4,8,16,32,64}, yticklabels={1,2,4,8,16,32,64}},
}
% \figThreadsBand: the hyperthreading (SMT) band at 64 threads, drawn under the data.
\newcommand{\figThreadsBand}{%
  \fill[figGrid!55] ({axis cs:45.25,1} |- {rel axis cs:0,0}) rectangle ({rel axis cs:1,1});
  \node[rotate=90, anchor=east, inner sep=1.5pt, font=\figPointFont, text=figInkTwo]
    at ({axis cs:96,1} |- {rel axis cs:0,1}) {Hyperthreading};}
% \figThreadsSeries{file}{column}: joining lines then marks, one per configuration.
\newcommand{\figThreadsSeries}[2]{%
  \pgfplotsinvokeforeach{rl1,phi,psi,dual,lcp}{%
    \addplot[ms link/##1, discard if not={key}{##1}] table[x=x, y=#2] {#1};}%
  \pgfplotsinvokeforeach{rl1,phi,psi,dual,lcp}{%
    \addplot[ms key/##1, discard if not={key}{##1}] table[x=x, y=#2] {#1};}}
\begin{tikzpicture}
\begin{axis}[name=p1, speed axis, xmode=log, log basis x=2, ymode=log, ylabel={\figSpeedLabel},
    fig group legend, legend to name=legThreadsMS]
  \figLegendGroupName{ms2:}\figLegendKey{lcp}{min}\figLegendKey{psi}{psi}\figLegendKey{phi}{phi}\figLegendKey{dual}{dual}

\begin{axis}[name=p2, speed axis, xmode=log, log basis x=2, ymode=log, yticklabels={}, at={($(p1.south east)+(\figHsep,0)$)},
    anchor=south west, fig group legend, legend to name=legThreadsRope]
  \figLegendGroupName{ropebwt3:}\figLegendKey{rl1}{-l1}
  \figThreadsBand
  \figThreadsSeries{data/fig-threads/human_ont_modern.dat}{y}
  \figPanelTitle{\figTitle{human_ont_modern}}
\end{axis}
\begin{axis}[name=p3, speedup axis, xmode=log, log basis x=2, ymode=log, log basis y=2, ylabel={Speedup over 1 thread},

  \figThreadsBand

  \figPanelTitle{\figTitle{human_short}}

\begin{axis}[name=p4, speedup axis, xmode=log, log basis x=2, ymode=log, log basis y=2, yticklabels={}, at={($(p3.south east)+(\figHsep,0)$)},
