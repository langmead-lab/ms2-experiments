% Refinements on top of figures-style.tex, used by the figures folded into the
% paper: panel titles in the lower right corner on a common baseline, larger
% marks and point labels, and legends split into outlined groups.
\renewcommand{\figPanelTitle}[1]{%
  \node[anchor=base east, inner sep=0pt, font=\figTitleFont, text=figInk]
    at ($(rel axis cs:1,0)+(-3pt,4.5pt)$) {#1};}
\renewcommand{\figPointFont}{\sffamily\fontsize{7.5}{8.5}\selectfont}
\pgfplotsset{
  every axis plot post/.append style={every mark/.append style={scale=1.3}},
  fig group legend/.style={fig legend,
    legend image post style={scale=1.35},
    legend style={draw=figInkTwo!60, rounded corners=1.5pt, line width=0.4pt,
      inner xsep=3pt, inner ysep=1.5pt}},
  % Narrower spacing between entries, for legend rows that would otherwise
  % run past the panels.
  fig tight group legend/.style={fig group legend,
    legend style={/tikz/every even column/.append style={column sep=3pt}}},
}

% \figLegendRow{first panel}{last panel}{legend name}{legend name}[...]: named
% legend boxes side by side, centered above a row of panels. The third box is
% optional (\figLegendRowThree).
\newcommand{\figLegendRow}[4]{%
  \node[anchor=south, inner sep=0pt]
    at ($(#1.north west)!0.5!(#2.north east)+(0,3pt)$)
    {\pgfplotslegendfromname{#3}\hspace{6pt}\pgfplotslegendfromname{#4}};}
\newcommand{\figLegendRowThree}[5]{%
  \node[anchor=south, inner sep=0pt]
    at ($(#1.north west)!0.5!(#2.north east)+(0,3pt)$)
    {\pgfplotslegendfromname{#3}\hspace{4pt}\pgfplotslegendfromname{#4}\hspace{4pt}\pgfplotslegendfromname{#5}};}
